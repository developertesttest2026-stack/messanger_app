async function loadNewMessages() {
    try {
        const currentRoom = sessionStorage.getItem("currentRoom") || "general";
        const response = await fetch(
            `http://localhost:8000/rooms/${currentRoom}/messages/poll?since=${lastMessageTime}`
        );
        
        if (response.ok) {
            const messages = await response.json();
            
            if (messages.length > 0) {
                // 🎯 Рендерим только новые сообщения
                messages.forEach(msg => renderSingleMessage(msg));
                
                // 🕒 Обновляем время последнего сообщения

// 🔧 Добавь этот код после рендера сообщений (например, в конце DOMContentLoaded):

chatWindow.addEventListener('click', async (e) => {
    // Проверяем, кликнули ли по кнопке редактирования
    if (e.target.classList.contains('edit-btn')) {
        const messageDiv = e.target.closest('.message');
        const messageId = messageDiv.dataset.messageId;
        const textElement = messageDiv.querySelector('.message-text');
        const currentText = textElement.innerText;
        
        // 🪄 Показываем prompt с текущим текстом
        const newText = prompt('Исправь сообщение:', currentText);
        
        // Если пользователь нажал "Отмена" или ввёл пустой текст — выходим
        if (newText === null || newText.trim() === '') return;
        
        try {
            // 📤 Отправляем PATCH-запрос на сервер
            const response = await fetch(`http://localhost:8000/messages/${messageId}`, {
                method: 'PATCH',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ text: newText.trim() })
            });
            
            if (response.ok) {
                // ✅ Обновляем текст в интерфейсе
                textElement.innerText = newText.trim();
                
                // 🔔 Можно добавить визуальный фидбек
                textElement.style.backgroundColor = '#fff3cd';
                setTimeout(() => {
                    textElement.style.backgroundColor = '';
                }, 1000);
            } else {
                const error = await response.json();
                alert(`Ошибка: ${error.detail || 'Не удалось обновить сообщение'}`);
            }
        } catch (error) {
            console.error('Ошибка при редактировании:', error);
            alert('Ошибка сети. Попробуй ещё раз.');
        }
    }
});

class TypingStatus(BaseModel):
    user_id: str
    room: str

# 🗄️ Временное хранилище (в памяти)
typing_indicators: Dict[str, datetime] = {}

# 📤 Эндпоинт: пользователь начал печатать
@app.post("/typing", status_code=200)
async def post_typing(status: TypingStatus):
    key = f"{status.user_id}:{status.room}"
    typing_indicators[key] = datetime.now()
    return {"status": "ok"}

# 📥 Эндпоинт: получить список печатающих в комнате
@app.get("/rooms/{room_name}/typing", response_model=List[str])
async def get_typing(room_name: str):
    now = datetime.now()
    typing_now = []
    
    # Проверяем всех, кто печатал в этой комнате
    for key, last_typed in list(typing_indicators.items()):
        user_id, room = key.split(":")
        if room == room_name and now - last_typed < timedelta(seconds=3):
            # Находим имя пользователя
            user = get_user_by_id(user_id)
            if user:
                typing_now.append(user['username'])
        else:
            # Удаляем устаревшие записи
            if room == room_name:
                del typing_indicators[key]
    
    return typing_now

- Использовать только CSS (без изменения HTML-структуры)
- Выделение не должно мешать чтению и быть доступным (контраст, не только цвет)

 Шаг 1: Добавляем класс к своему имени в `renderUsers`
javascript
function renderUsers(data) {
    userList.innerHTML = '';
    const currentUserId = sessionStorage.getItem("currentUser");
    
    data.forEach(u => {
        const li = document.createElement("li");
        li.innerText = u.name;
        
        // 🌟 Добавляем класс, если это текущий пользователь
        if (u.id === currentUserId) {
            li.classList.add('current-user');
        }
        
        userList.appendChild(li);
    });
}
